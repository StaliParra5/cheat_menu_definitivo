	# COMENTARIO ESTRATÉGICO: Incrementa en 5 el atributo base de Diplomacia.

	add_diplomacy = 5
}

cmd_effect_add_renown_1k = {
	# COMENTARIO ESTRATÉGICO: Añade 1,000 de renombre de dinastía.
	if = {
		limit = { exists = dynasty }
		dynasty = { add_dynasty_renown = 1000 }
	}
}

# --- ASIGNACIÓN RÁPIDA DE TRAITS POPULARES ---
cmd_effect_toggle_genius = {
	# COMENTARIO ESTRATÉGICO: Alterna el rasgo Genio (intelligence_3).
	if = {
		limit = { has_trait = intelligence_3 }
		remove_trait = intelligence_3
	}
	else = {
		add_trait = intelligence_3
	}
}
