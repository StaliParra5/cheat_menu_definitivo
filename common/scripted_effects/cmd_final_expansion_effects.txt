# Strategic Comment: Efectos scriptados para la expansión final (Ejércitos Especiales y Rasgos Genéticos).

# === 1. INVOCACIÓN DE EJÉRCITOS ESPECIALES ===

cmd_effect_spawn_special_troops = {
	# COMENTARIO ESTRATÉGICO: Invoca 5,000 tropas de infantería pesada especiales en la capital del jugador humano.
	spawn_army = {
		name = "Guardia de Honor Legendaria"
		levies = 0
		men_at_arms = {
			type = heavy_infantry
			stacks = 50
		}
		location = scope:actor.capital_province
	}
}

# === 2. PANEL DE GENÉTICA Y RASGOS RAROS ===

cmd_effect_toggle_pure_blood = {
	# COMENTARIO ESTRATÉGICO: Alterna el rasgo genético Sangre Pura (pure_blood).
	if = {
		limit = { has_trait = pure_blood }
		remove_trait = pure_blood
	}
	else = {
		add_trait = pure_blood
	}
}
