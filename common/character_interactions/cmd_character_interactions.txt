# Strategic Comment: Interacciones de personaje (Clic Derecho) para el Cheat Menu Definitivo.
# Permite al jugador interactuar con cualquier personaje del mapa para ejecutar efectos inmediatos.

# --- Truco: Aumentar Límite de Dominio y Ducados (+100) ---
cmd_interaction_domain_limit = {
	category = interaction_category_diplomacy
	common_interaction_options = yes

	is_shown = {
		scope:actor = { is_ai = no }
	}

	on_accept = {
		scope:recipient = {
			cmd_effect_toggle_domain_limit = yes
		}
	}
}
