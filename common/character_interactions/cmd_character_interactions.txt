# Strategic Comment: Interacciones de personaje (Clic Derecho) para el Cheat Menu Definitivo.
# Con common_interaction = yes y auto_accept = yes para visibilidad directa y ejecución instantánea.

# --- Límite de Dominio ---
cmd_interaction_domain_limit = {
	category = interaction_category_friendly
	common_interaction = yes
	auto_accept = yes
	use_diplomatic_range = no

	is_shown = {
		scope:actor = { is_ai = no }
	}

	on_accept = {
		scope:recipient = {
			cmd_effect_toggle_domain_limit = yes
		}
	}
}
